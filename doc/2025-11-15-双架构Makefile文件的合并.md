# 将两种架构的Makefile文件进行合并

## 通过定义参数ARCH从而实现不同架构的编译以及运行

ARCH?=riscv
设定ARCH的默认值为riscv，进而保证直接运行make qemu会自动启动riscv操作系统，与初始项目保持一致，进而便于其他组员协作编程

## 通过定义参数A接收ARCH的值，进而实现不同架构的编译规则统一

A=$(ARCH)
对于原始makefile，.c文件的编译规则无法统一，riscv中要从riscv/kernel编译至kernel，loongarch要从loongarch/kernel编译至kernel,通过这种方式可以通过同一语句实现两种规则进行编译，减少了代码量

## 对于user及kernel的编译依赖不同，采用user_deps和kernel_deps来进行统一

KERNEL_DEPS = $(OBJS)  $(AR)/$A/$K/kernel.ld
USER_DEPS = $U/%.o $(ULIB)
ifeq ($(ARCH),loongarch)
KERNEL_DEPS += $U/initcode
USER_WAYS = $(LD) $(LDFLAGS) -N -e main -Ttext 0 -o $@ $^
endif
ifeq ($(ARCH),riscv)
USER_WAYS = $(LD) $(LDFLAGS) -T  $(AR)/$A/$U/user.ld -o $@ $< $(ULIB)
USER_DEPS +=  $(AR)/$A/$U/user.ld
endif

## 目标OBJ以及UPROSGS进行合并

首先添加架构公有的目标，随后采用条件编程添加特有的目标

OBJS = \
  $K/entry.o \
  $K/start.o \
  $K/console.o \
  $K/printf.o \
  $K/main.o \
  ...

ifeq ($(ARCH),riscv)
OBJS += \
  $K/start.o \
  $K/trampoline.o \
  $K/plic.o \
  $K/virtio_disk.o
endif
ifeq ($(ARCH),loongarch)
OBJS+=\
  $K/tlbrefill.o\
  $K/merror.o\
  $K/apic.o\
  $K/extioi.o\
  $K/ramdisk.o\
  $K/uservec.o\
endif
UPROGS=\
	$U/_cat\
	$U/_echo\
	$U/_forktest\
	$U/_grep\
    ...

ifeq ($(ARCH),riscv)
UPROGS += $U/_grind\
	$U/_logstress\
	$U/_forphan\
	$U/_dorphan\
	...
endif

## 为loongarch架构编译qemu规则

原始loongarch项目没有qemu规则，只有all规则，并且采用特定的qemu虚拟机，而该虚拟机版本过低，存在不适配高版本ubuntu的问题，因此编写了以下qemu运行规则，使得编译与运行更加便捷
QEMU = qemu-system-loongarch64
QEMU_OPTS = -kernel kernel/kernel -m 1G -nographic -smp 1
QEMU_OPTS += -drive file=fs.img,if=none,format=raw,id=x0
QEMU_OPTS += -device virtio-blk-pci,drive=x0
QEMU_OPTS += -no-reboot
QEMU_OPTS += -device virtio-net-pci,netdev=net0
QEMU_OPTS += -netdev user,id=net0
QEMU_OPTS += -rtc base=utc

all: fs.img $K/kernel 

qemu: all
	$(QEMU) $(QEMU_OPTS)

## 为loongarch架构交叉编译器的配置创建特有规则

ifeq ($(ARCH),loongarch)
export CC_PREFIX = /opt/cross-tools
export PATH := $(CC_PREFIX)/bin:$(PATH)
export LD_LIBRARY_PATH := $(CC_PREFIX)/lib:$(CC_PREFIX)/loongarch64-unknown-linux-gnu/lib:$(LD_LIBRARY_PATH)
endif

原loongarch项目中每次编译都必须对编译器进行配置，因此将该配置规则写入makefile，便于调试

# 其他优化

$U/initcode
	$(LD) $(LDFLAGS) -T $A/$K/kernel.ld -o $K/kernel $(OBJS)
	$(OBJDUMP) -S $K/kernel > $K/kernel.asm
	$(OBJDUMP) -t $K/kernel | sed '1,/SYMBOL TABLE/d; s/ .* / /; /^$$/d' > $K/kernel.sym
将编译产生的中间文件放入KERNEL及USER目录下，使得结构更加清晰
